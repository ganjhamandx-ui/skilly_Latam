class IniciarPagoTransferenciaEmpresaView(APIView):
    authentication_classes = []
    permission_classes = [AllowAny]
    def post(self, request, referencia):
        datos_referencia = leer_referencia_seguimiento(
            referencia
        )
        if datos_referencia is None:
            return Response(
                {
                    "mensaje": (
                        "La referencia de seguimiento "
                        "no es válida."
                    )
                },
                status=status.HTTP_400_BAD_REQUEST,
            )
        entrada_serializer = (
            IniciarPagoTransferenciaSerializer(
                data=request.data
            )
        )
        if not entrada_serializer.is_valid():
            return Response(
                entrada_serializer.errors,
                status=status.HTTP_400_BAD_REQUEST,
            )
        datos = entrada_serializer.validated_data
        comprobante = datos[
            "comprobante_transferencia"
        ]
        referencia_transaccion = datos.get(
            "referencia_transaccion"
        )
        if referencia_transaccion == "":
            referencia_transaccion = None
        with transaction.atomic():
            prestacion = (
                PrestacionServicio.objects
                .select_for_update()
                .select_related(
                    "id_solicitud",
                    "id_profesional",
                )
                .filter(
                    id_solicitud_id=(
                        datos_referencia[
                            "id_solicitud"
                        ]
                    ),
                    id_solicitud__id_contacto_id=(
                        datos_referencia[
                            "id_contacto"
                        ]
                    ),
                )
                .first()
            )
            if prestacion is None:
                return Response(
                    {
                        "mensaje": (
                            "No existe una prestación "
                            "autorizada para esta referencia."
                        )
                    },
                    status=status.HTTP_404_NOT_FOUND,
                )
            pago_existente = (
                Pago.objects
                .select_for_update()
                .filter(
                    id_prestacion=prestacion
                )
                .first()
            )
            if pago_existente is not None:
                if (
                    pago_existente.metodo_pago
                    != "TRANSFERENCIA"
                ):
                    return Response(
                        {
                            "mensaje": (
                                "La prestación ya posee "
                                "un pago iniciado con otro "
                                "medio de pago."
                            )
                        },
                        status=status.HTTP_409_CONFLICT,
                    )
                pago_data = dict(
                    PagoEstadoComunicableSerializer(
                        pago_existente
                    ).data
                )
                pago_data["metodo_pago"] = (
                    pago_existente.metodo_pago
                )
                pago_data["estado_verificacion"] = (
                    pago_existente.estado_verificacion
                )
                return Response(
                    {
                        "mensaje": (
                            "La transferencia ya se "
                            "encontraba registrada."
                        ),
                        "pago": pago_data,
                    },
                    status=status.HTTP_200_OK,
                )
            if (
                prestacion.estado
                != "PENDIENTE_CONTRATACION"
            ):
                return Response(
                    {
                        "mensaje": (
                            "La prestación no se encuentra "
                            "pendiente de contratación."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            condiciones = str(
                prestacion.condiciones_comunicadas
                or ""
            ).strip()
            if not condiciones:
                return Response(
                    {
                        "mensaje": (
                            "Skilly debe comunicar las "
                            "condiciones de la prestación "
                            "antes de iniciar el pago."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            servicio = (
                prestacion.id_solicitud.id_servicio
            )
            if servicio is None:
                return Response(
                    {
                        "mensaje": (
                            "La solicitud no posee un "
                            "servicio asociado para "
                            "determinar el precio."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if (
                servicio.id_profesional_id
                != prestacion.id_profesional_id
            ):
                return Response(
                    {
                        "mensaje": (
                            "El profesional de la prestación "
                            "no coincide con el profesional "
                            "del servicio."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            reserva = (
                ReservaDisponibilidad.objects
                .select_for_update()
                .filter(
                    id_solicitud=(
                        prestacion.id_solicitud
                    ),
                    estado="PENDIENTE_PAGO",
                )
                .order_by(
                    "-fecha_reserva",
                    "-id_reserva",
                )
                .first()
            )
            if reserva is None:
                return Response(
                    {
                        "mensaje": (
                            "La solicitud no posee un horario "
                            "seleccionado pendiente de pago."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            bloque = (
                Disponibilidad.objects
                .select_for_update()
                .filter(
                    id_disponibilidad=(
                        reserva.id_disponibilidad_id
                    )
                )
                .first()
            )
            if bloque is None:
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado ya no "
                            "se encuentra disponible."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if (
                bloque.id_profesional_id
                != prestacion.id_profesional_id
            ):
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado no "
                            "corresponde al profesional "
                            "de la prestación."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if bloque.estado != "DISPONIBLE":
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado ya no "
                            "está disponible. Debe elegir "
                            "otro horario antes de pagar."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            reserva_activa_otra = (
                ReservaDisponibilidad.objects
                .filter(
                    id_disponibilidad=bloque,
                    estado="ACTIVA",
                )
                .exists()
            )
            if reserva_activa_otra:
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado ya fue "
                            "reservado. Debe elegir otro "
                            "horario antes de pagar."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            inicio_bloque = timezone.make_aware(
                datetime.combine(
                    bloque.fecha,
                    bloque.hora_inicio,
                ),
                timezone.get_current_timezone(),
            )
            if inicio_bloque <= timezone.now():
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado ya "
                            "comenzó o quedó en el pasado. "
                            "Debe elegir otro horario."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            try:
                precio_normalizado = (
                    normalizar_precio_clp(
                        servicio.referencia_precio,
                        permitir_vacio=False,
                    )
                )
            except ValidationError:
                return Response(
                    {
                        "mensaje": (
                            "El servicio no posee un precio "
                            "final válido para iniciar "
                            "el pago."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            partes_precio = (
                precio_normalizado.split()
            )
            if len(partes_precio) != 2:
                return Response(
                    {
                        "mensaje": (
                            "El precio final del servicio "
                            "no pudo interpretarse."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            try:
                monto_total = Decimal(
                    partes_precio[0]
                ).quantize(
                    Decimal("0.01"),
                    rounding=ROUND_HALF_UP,
                )
            except InvalidOperation:
                return Response(
                    {
                        "mensaje": (
                            "El monto final del servicio "
                            "no es válido."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            moneda = partes_precio[1]
            parametro_comision = (
                ParametroSistema.objects
                .filter(
                    clave="comision_porcentaje"
                )
                .first()
            )
            if parametro_comision is None:
                return Response(
                    {
                        "mensaje": (
                            "La comisión del sistema no se "
                            "encuentra configurada."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            try:
                porcentaje_comision = Decimal(
                    str(
                        parametro_comision.valor_actual
                    ).strip()
                )
            except (
                InvalidOperation,
                TypeError,
                ValueError,
            ):
                return Response(
                    {
                        "mensaje": (
                            "La configuración de comisión "
                            "del sistema no es válida."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if (
                not porcentaje_comision.is_finite()
                or porcentaje_comision < Decimal("0")
                or porcentaje_comision > Decimal("100")
            ):
                return Response(
                    {
                        "mensaje": (
                            "La configuración de comisión "
                            "del sistema está fuera del "
                            "rango permitido."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            porcentaje_comision = (
                porcentaje_comision.quantize(
                    Decimal("0.0001"),
                    rounding=ROUND_HALF_UP,
                )
            )
            monto_comision = (
                monto_total
                * porcentaje_comision
                / Decimal("100")
            ).quantize(
                Decimal("0.01"),
                rounding=ROUND_HALF_UP,
            )
            monto_neto_profesional = (
                monto_total - monto_comision
            ).quantize(
                Decimal("0.01"),
                rounding=ROUND_HALF_UP,
            )
            ahora = timezone.now()
            pago = Pago.objects.create(
                id_prestacion=prestacion,
                monto_total=monto_total,
                moneda=moneda,
                porcentaje_comision_aplicado=(
                    porcentaje_comision
                ),
                monto_comision=monto_comision,
                monto_neto_profesional=(
                    monto_neto_profesional
                ),
                estado_pago=(
                    "PENDIENTE_VERIFICACION"
                ),
                metodo_pago="TRANSFERENCIA",
                proveedor_pago=None,
                estado_verificacion="PENDIENTE",
                comprobante_transferencia=(
                    comprobante
                ),
                fecha_pago=None,
                fecha_retencion=None,
                fecha_liberacion=None,
                referencia_transaccion=(
                    referencia_transaccion
                ),
            )
            HistorialEstadoPago.objects.create(
                id_pago=pago,
                id_usuario_skilly=None,
                estado_anterior=None,
                estado_nuevo=(
                    "PENDIENTE_VERIFICACION"
                ),
                fecha_cambio=ahora,
                observacion=(
                    "Transferencia informada por la "
                    "empresa y pendiente de verificación "
                    "por Skilly."
                ),
            )
            EventoTrazabilidad.objects.create(
                entidad_tipo="pago",
                entidad_id=pago.id_pago,
                accion="INICIAR_PAGO_TRANSFERENCIA",
                detalle_resumido=(
                    f"Pago {pago.id_pago} iniciado para "
                    f"prestación "
                    f"{prestacion.id_prestacion}. "
                    f"Estado: "
                    f"PENDIENTE_VERIFICACION."
                )[:255],
                fecha_evento=ahora,
            )
        pago_data = dict(
            PagoEstadoComunicableSerializer(
                pago
            ).data
        )
        pago_data["metodo_pago"] = (
            pago.metodo_pago
        )
        pago_data["estado_verificacion"] = (
            pago.estado_verificacion
        )
        return Response(
            {
                "mensaje": (
                    "Transferencia registrada. "
                    "El pago quedó pendiente de "
                    "verificación por Skilly."
                ),
                "pago": pago_data,
                "reserva_confirmada": False,
                "prestacion_contratada": False,
            },
            status=status.HTTP_201_CREATED,
        )

class DetallePagoAdministracionView(APIView):
    authentication_classes = [
        SesionSkillyAuthentication
    ]
    permission_classes = [
        EsAdministracionSkilly
    ]
    def get(self, request, id_pago):
        pago = (
            Pago.objects
            .select_related(
                "id_prestacion",
            )
            .filter(
                id_pago=id_pago
            )
            .first()
        )
        if pago is None:
            return Response(
                {
                    "mensaje": (
                        "El pago solicitado no existe."
                    )
                },
                status=status.HTTP_404_NOT_FOUND,
            )
        historial = (
            HistorialEstadoPago.objects
            .select_related(
                "id_usuario_skilly",
            )
            .filter(
                id_pago=pago
            )
            .order_by(
                "fecha_cambio",
                "id_historial_pago",
            )
        )
        return Response(
            {
                "pago": (
                    PagoAdministracionSerializer(
                        pago
                    ).data
                ),
                "historial": (
                    HistorialEstadoPagoAdministracionSerializer(
                        historial,
                        many=True,
                    ).data
                ),
            },
            status=status.HTTP_200_OK,
        )
    def post(self, request, id_pago):
        entrada_serializer = (
            VerificarPagoTransferenciaSerializer(
                data=request.data
            )
        )
        if not entrada_serializer.is_valid():
            return Response(
                entrada_serializer.errors,
                status=status.HTTP_400_BAD_REQUEST,
            )
        observacion = (
            entrada_serializer.validated_data.get(
                "observacion"
            )
        )
        if observacion is not None:
            observacion = observacion.strip()
        if not observacion:
            observacion = (
                "Transferencia verificada por Skilly."
            )
        with transaction.atomic():
            pago = (
                Pago.objects
                .select_for_update()
                .filter(
                    id_pago=id_pago
                )
                .first()
            )
            if pago is None:
                return Response(
                    {
                        "mensaje": (
                            "El pago solicitado no existe."
                        )
                    },
                    status=status.HTTP_404_NOT_FOUND,
                )
            if pago.metodo_pago != "TRANSFERENCIA":
                return Response(
                    {
                        "mensaje": (
                            "Esta operación solo permite "
                            "verificar pagos realizados "
                            "mediante transferencia."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if (
                pago.estado_pago == "RETENIDO"
                and pago.estado_verificacion
                == "VERIFICADO"
            ):
                return Response(
                    {
                        "mensaje": (
                            "La transferencia ya se "
                            "encontraba verificada."
                        ),
                        "pago": (
                            PagoAdministracionSerializer(
                                pago
                            ).data
                        ),
                    },
                    status=status.HTTP_200_OK,
                )
            if (
                pago.estado_pago
                != "PENDIENTE_VERIFICACION"
                or pago.estado_verificacion
                != "PENDIENTE"
            ):
                return Response(
                    {
                        "mensaje": (
                            "El pago no se encuentra en "
                            "un estado válido para verificar "
                            "la transferencia."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            prestacion = (
                PrestacionServicio.objects
                .select_for_update()
                .filter(
                    id_prestacion=(
                        pago.id_prestacion_id
                    )
                )
                .first()
            )
            if prestacion is None:
                return Response(
                    {
                        "mensaje": (
                            "La prestación asociada al pago "
                            "no existe."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if (
                prestacion.estado
                != "PENDIENTE_CONTRATACION"
            ):
                return Response(
                    {
                        "mensaje": (
                            "La prestación no se encuentra "
                            "pendiente de contratación."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            reserva = (
                ReservaDisponibilidad.objects
                .select_for_update()
                .filter(
                    id_solicitud_id=(
                        prestacion.id_solicitud_id
                    ),
                    estado="PENDIENTE_PAGO",
                )
                .order_by(
                    "-fecha_reserva",
                    "-id_reserva",
                )
                .first()
            )
            if reserva is None:
                return Response(
                    {
                        "mensaje": (
                            "No existe una selección de "
                            "horario pendiente de pago "
                            "para esta prestación."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            bloque = (
                Disponibilidad.objects
                .select_for_update()
                .filter(
                    id_disponibilidad=(
                        reserva.id_disponibilidad_id
                    )
                )
                .first()
            )
            if bloque is None:
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado ya no "
                            "existe."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if (
                bloque.id_profesional_id
                != prestacion.id_profesional_id
            ):
                return Response(
                    {
                        "mensaje": (
                            "El horario seleccionado no "
                            "corresponde al profesional "
                            "de la prestación."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )
            if bloque.estado != "DISPONIBLE":
                return Response(
                    {
                        "mensaje": (

function formatoFecha(valor) {
  if (!valor) return 'No registrada'

  const fecha = new Date(valor)

  if (Number.isNaN(fecha.getTime())) {
    return String(valor)
  }

  return fecha.toLocaleString('es-CL', {
    dateStyle: 'medium',
    timeStyle: 'short',
  })
}

function obtenerMensaje(datos, codigo) {
  if (typeof datos?.mensaje === 'string') {
    return datos.mensaje
  }

  if (typeof datos?.detail === 'string') {
    return datos.detail
  }

  if (datos && typeof datos === 'object') {
    const errores = Object.entries(datos).flatMap(
      ([campo, valor]) =>
        Array.isArray(valor)
          ? valor.map((mensaje) => `${campo}: ${mensaje}`)
          : [],
    )

    if (errores.length) {
      return errores.join(' | ')
    }
  }

  return `Operación no completada (HTTP ${codigo}).`
}

    const pago = detallePago?.pago
    if (!pago || verificandoPago || !recepcionComprobada) return

    const pendiente =
      pago.metodo_pago === 'TRANSFERENCIA' &&
      pago.estado_pago === 'PENDIENTE_VERIFICACION' &&
      pago.estado_verificacion === 'PENDIENTE'

    if (!pendiente) {
      setErrorDetallePago('El pago no admite verificación.')
      return
    }

    if (!observacionVerificacion.trim()) {
      setErrorDetallePago('Debes registrar una observación administrativa.')
      return
    }

    const confirmado = window.confirm(
      `Confirma que Skilly comprobó la recepción efectiva ` +
      `del dinero del pago #${pago.id_pago}. ` +
      'Esta operación retendrá el pago y activará la reserva. ' +
      'No contratará automáticamente la prestación.'
    )

    if (!confirmado) return

    setVerificandoPago(true)
    setErrorDetallePago('')
    setAvisoPago('')

    try {
      const token = await obtenerCsrf()

      const datos = await obtenerJson(
        `/pagos/administracion/${pago.id_pago}/`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': token,
          },
          body: JSON.stringify({
            accion: 'VERIFICAR_TRANSFERENCIA',
            observacion: observacionVerificacion.trim(),
          }),
        },
      )

      if (
        datos.pago?.estado_pago !== 'RETENIDO' ||
        datos.pago?.estado_verificacion !== 'VERIFICADO'
      ) {
        throw new Error('No se confirmó el estado del pago.')
      }

      const actualizado = await obtenerJson(
        `/pagos/administracion/${pago.id_pago}/`,
      )

      setDetallePago(actualizado)
      setRecepcionComprobada(false)
      setObservacionVerificacion('')
      setAvisoPago('Transferencia verificada administrativamente.')
      await cargarPagos()
    } catch (fallo) {
      setErrorDetallePago(fallo.message)
    } finally {
      setVerificandoPago(false)
    }
  }

  async function consultarDetallePago(idPago) {
    if (!Number.isSafeInteger(idPago) || idPago <= 0) {
      setErrorDetallePago('Identificador de pago no valido.')
      return
    }

    setRecepcionComprobada(false)
    setObservacionVerificacion('')
    setAvisoPago('')
    setCargandoDetallePago(true)
    setErrorDetallePago('')
    setDetallePago(null)

    try {
      const datos = await obtenerJson(
        `/pagos/administracion/${idPago}/`,
      )

      if (!datos.pago || !Array.isArray(datos.historial)) {
        throw new Error('La respuesta del detalle de pago no es valida.')
      }

      setDetallePago(datos)
    } catch (fallo) {
      setErrorDetallePago(fallo.message)
    } finally {
      setCargandoDetallePago(false)
    }
  }


  async function agendarPrimeraReunion() {
    const prestacion = coordinacion?.prestacion

    if (!seleccion || !prestacion || agendandoPrimeraReunion) {
      return
    }

    const pagoValido = pagos.some((pago) =>
      Number(pago.id_prestacion) === Number(prestacion.id_prestacion) &&
      pago.estado_pago === 'RETENIDO' &&
      pago.estado_verificacion === 'VERIFICADO'
    )

    const reservaValida = (coordinacion.reservas ?? []).some(
      (reserva) =>
        reserva.estado === 'ACTIVA' &&
        reserva.disponibilidad?.estado === 'RESERVADA'
    )

    const reunionExistente = (coordinacion.reuniones ?? []).some(
      (reunion) => reunion.estado === 'AGENDADA'
    )

    if (
      prestacion.estado !== 'PENDIENTE_CONTRATACION' ||
      !pagoValido ||
      !reservaValida ||
      reunionExistente
    ) {
      setErrorCoordinacion(
        'No se cumplen las condiciones para agendar la primera reunión.'
      )
      return
    }

    const confirmado = window.confirm(
      `¿Confirmas agendar la primera reunión de la ` +
      `prestación #${prestacion.id_prestacion}? ` +
      'Se utilizará el horario ya reservado. ' +
      'Si el backend valida todas las condiciones, ' +
      'la prestación pasará a CONTRATADA.'
    )

    if (!confirmado) return

    setAgendandoPrimeraReunion(true)
    setErrorCoordinacion('')
    setAviso('')

    try {
      const csrfToken = await obtenerCsrf()

      const resultado = await obtenerJson(
        `/coordinacion/administracion/prestaciones/${prestacion.id_prestacion}/agenda/`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': csrfToken,
          },
          body: JSON.stringify({
            accion: 'AGENDAR_PRIMERA_REUNION',
          }),
        },
      )

      if (
        resultado.prestacion_contratada !== true ||
        resultado.prestacion?.estado !== 'CONTRATADA' ||
        resultado.reunion?.estado !== 'AGENDADA'
      ) {
        throw new Error(
          'La respuesta requiere revisión administrativa.'
        )
      }

      const actualizado = await obtenerJson(
        `/coordinacion/administracion/solicitudes/${seleccion.id_solicitud}/coordinacion/`,
      )

      setCoordinacion(actualizado)
      setAviso(
        'Primera reunión agendada. Prestación CONTRATADA.'
      )
    } catch (fallo) {
      setErrorCoordinacion(fallo.message)
    } finally {
      setAgendandoPrimeraReunion(false)
    }
  }

  async function iniciarSesion(evento) {
    evento.preventDefault()
    setCargando(true)
    setError('')
    setAviso('')
    try {
      const csrfToken = await obtenerCsrf()
      const datos = await obtenerJson('/acceso/iniciar-sesion/', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-CSRFToken': csrfToken,
        },
        body: JSON.stringify({
          correo: correo.trim(),
          contrasena,
        }),
      })
      setContrasena('')
      if (datos.usuario?.rol !== 'ADMIN_SKILLY') {
        throw new Error('Esta cuenta no tiene permisos de administración Skilly.')
      }
      setUsuario(datos.usuario)
      setAviso('Sesión administrativa iniciada.')
      await cargar()
    } catch (fallo) {
      setContrasena('')
      setError(fallo.message)
    } finally {
      setCargando(false)
    }
  }

  async function autoasignar(solicitud) {
    if (asignando || !solicitud) return

    const id = solicitud.id_solicitud

    if (!window.confirm(
      `?Confirmas que deseas asumir la solicitud #${id} como responsable interno?`
    )) return

    setAsignando(true)
    setError('')
    setAviso('')

    try {
      const csrfToken = await obtenerCsrf()
      const datos = await obtenerJson(
        `/coordinacion/administracion/solicitudes/${id}/asignacion/`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': csrfToken,
          },
          body: JSON.stringify({ modo: 'AUTOASIGNAR' }),
        },
      )

      if (!datos.asignacion || datos.asignacion.estado !== 'ACTIVA') {
        throw new Error('La API no confirmó la asignación.')
      }

      setAsignaciones((anteriores) => ({
        ...anteriores,
        [id]: {
          actual: datos.asignacion,
          respuesta: datos,
        },
      }))

      setAviso(`Solicitud #${id} asignada correctamente.`)
      setPestana('gestion')
    } catch (fallo) {
      setError(fallo.message)
    } finally {
      setAsignando(false)
    }
  }

                        <div className="adm-aviso" role="status">
                          {avisoPago}
                        </div>
                      )}

                      <h4>Historial de estados</h4>
                      {detallePago.historial.length === 0 ? (
                        <p className="adm-vacio">
                          No existen cambios de estado registrados.
                        </p>
                      ) : (
                        <div className="adm-lista">
                          {detallePago.historial.map((evento) => (
                            <div
                              key={evento.id_historial_pago}
                              className="adm-item"
                            >
                              <span className="adm-item-cuerpo">
                                <strong>
                                  {evento.estado_anterior || 'INICIAL'}
                                  {' | '}
                                  {evento.estado_nuevo}
                                </strong>
                                <span>
                                  {evento.fecha_cambio
                                    ? new Date(
                                      evento.fecha_cambio,
                                    ).toLocaleString('es-CL')
                                    : 'Sin fecha'}
                                </span>
                                <span>
                                  {corregirTextoHistorico(evento.observacion) || 'Sin observaciones'}
                                </span>
                              </span>
                            </div>
                          ))}
                        </div>
                      )}
                    </>
                  )}
                </div>
                </VentanaAdministracion>
              )}
            </section>

            <section className="adm-panel">
              <div className="adm-panel-header">
                <div>
                  <span className="adm-eyebrow">BANDEJA ADMINISTRATIVA</span>
                  <h2>Solicitudes empresariales</h2>
                </div>
                <button type="button" className="adm-secundario"
                  onClick={cargar} disabled={cargando}>
                  {cargando ? 'Consultando...' : '↻ Recargar bandeja'}
                </button>
              </div>

              <div className="adm-pestanas">
                {[
                  ['sin', 'Sin asignar'],
                  ['mias', 'Asignadas a mí'],
                  ['gestion', 'En gestión'],
                  ['cerradas', 'Resueltas / cerradas'],
                  ['todas', 'Todas'],
                ].map(([clave, titulo]) => (
                  <button key={clave} type="button"
                    className={pestana === clave ? 'activa' : ''}
                    onClick={() => setPestana(clave)}>
                    {titulo}
                  </button>
                ))}
              </div>

              <input className="adm-busqueda" value={busqueda}
                onChange={(e) => setBusqueda(e.target.value)}
                placeholder="Buscar empresa, servicio, objetivo o estado..."
                aria-label="Buscar solicitudes" />

              {visibles.length === 0 ? (
                <p className="adm-vacio">No hay solicitudes en esta vista.</p>
              ) : (
                <div className="adm-lista">
                  {visibles.map((item) => (
                    <button type="button" className="adm-item"
                      key={item.id_solicitud}
                      onClick={() => setSeleccion(item)}>
                      <span className="adm-item-cuerpo">
                        <strong>#{item.id_solicitud} · {item.empresa?.nombre_organizacion || 'Empresa'}</strong>
                        <small>{item.servicio?.nombre || 'Servicio sin especificar'}</small>
                        <small>{item.objetivo}</small>
                        {asignaciones[item.id_solicitud]?.error
                          ? <small>Asignación: no disponible</small>
                          : <small>Responsable: {responsable(item)?.correo || 'Sin asignar'}</small>}
                      </span>
                      <span className="adm-estado">{item.estado}</span>
                    </button>
                  ))}
                </div>
              )}
            </section>

            {seleccion && (
              <VentanaAdministracion abierta titulo={`Solicitud empresarial #${seleccion.id_solicitud}`}
                bloqueado={asignando || proponiendo || confirmandoAsociacion || preparandoPrestacion || agendandoPrimeraReunion}
                onCerrar={() => setSeleccion(null)}>
              <section className="adm-panel adm-detalle">
                <div className="adm-panel-header">
                  <div>
                    <span className="adm-eyebrow">DETALLE EMPRESARIAL</span>
                    <h2>Solicitud #{seleccion.id_solicitud}</h2>
                  </div>
                  <button className="adm-secundario" type="button"
                    onClick={() => setSeleccion(null)}>Cerrar detalle</button>
                </div>
                <div className="adm-datos">
                  <div><small>Empresa</small><strong>{seleccion.empresa?.nombre_organizacion || 'No indicada'}</strong></div>
                  <div><small>Servicio</small><strong>{seleccion.servicio?.nombre || 'No indicado'}</strong></div>
                  <div><small>Contacto</small><strong>{seleccion.contacto?.nombre || 'No indicado'}</strong></div>
                  <div><small>Correo</small><strong>{seleccion.contacto?.correo || 'No indicado'}</strong></div>
                  <div><small>Estado</small><strong>{seleccion.estado}</strong></div>
                  <div><small>Responsable interno</small><strong>{responsable(seleccion)?.correo || 'Sin asignar'}</strong></div>
                </div>
                <section className="adm-coordinacion">
                  <h3>Coordinación y contratación</h3>
                  <p>Consulta administrativa del avance de esta solicitud.</p>

                  {cargandoCoordinacion && (
                    <p>Consultando coordinación...</p>
                  )}

                  {errorCoordinacion && (
                    <p role="alert">
                      No se pudo consultar la coordinación: {errorCoordinacion}
                    </p>
                  )}

                  {coordinacion && (
                    <>
                      <div className="adm-datos">
                        <div>
                          <small>Profesionales asociados</small>
                          <strong>{coordinacion.asociaciones?.length ?? 0}</strong>
                        </div>
                        <div>
                          <small>Reservas registradas</small>
                          <strong>{coordinacion.reservas?.length ?? 0}</strong>
                        </div>
                        <div>
                          <small>Reuniones registradas</small>
                          <strong>{coordinacion.reuniones?.length ?? 0}</strong>
                        </div>
                        <div>
                          <small>Eventos de coordinación</small>
                          <strong>{coordinacion.historial_coordinacion?.length ?? 0}</strong>
                        </div>
                        <div>
                          <small>Prestación</small>
                          <strong>
                            {coordinacion.prestacion?.estado || 'No preparada'}
                          </strong>
                        </div>
                      </div>


                      {seleccion.estado === 'RECIBIDA' &&
                        (coordinacion.asociaciones?.length ?? 0) === 0 &&
                        (coordinacion.reservas ?? []).some(
                          (reserva) => (
                            reserva.estado === 'PENDIENTE_PAGO' ||
                            reserva.estado === 'ACTIVA'
                          ),
                        ) &&
                        !coordinacion.prestacion && (
                          <button
                            type="button"
                            className="adm-accion"
                            disabled={proponiendo}
                            onClick={proponerAsociacion}
                          >
                            {proponiendo
                              ? 'Registrando propuesta...'
                              : 'Proponer asociación'}
                          </button>
                        )}

                      {(coordinacion.asociaciones ?? []).map(
                        (asociacion, indice) => (
                          <div key={indice}>
                            <p>
                              Profesional asociado:
                              {' '}{asociacion.estado}
                            </p>
                            {asociacion.estado === 'ACEPTADO' &&
                              !coordinacion.prestacion && (
                                <button
                                  type="button"
                                  className="adm-accion"
                                  disabled={preparandoPrestacion}
                                  onClick={() => prepararPrestacion(
                                    asociacion.profesional?.id_profesional,
                                  )}
                                >
                                  {preparandoPrestacion
                                    ? 'Preparando...'
                                    : 'Preparar prestación'}
                                </button>
                              )}
                            {asociacion.estado === 'PENDIENTE' &&
                              asignaciones[seleccion.id_solicitud]?.actual &&
                              !coordinacion.prestacion && (
                                <button
                                  type="button"
                                  className="adm-accion"
                                  disabled={confirmandoAsociacion}
                                  onClick={() => confirmarAsociacion(
                                    asociacion.profesional?.id_profesional,
                                  )}
                                >
                                  {confirmandoAsociacion
                                    ? 'Confirmando...'
                                    : 'Confirmar asociación'}
                                </button>
                              )}
                          </div>
                        ),
                      )}


                      {coordinacion.prestacion && (
                        <ComunicarCondicionesAdministracion
                          key={coordinacion.prestacion.id_prestacion}
                          idSolicitud={seleccion.id_solicitud}
                          prestacion={coordinacion.prestacion}
                          onRegistrado={(prestacionActualizada) => {
                            setCoordinacion((anterior) =>
                              anterior?.prestacion?.id_prestacion ===
                              prestacionActualizada.id_prestacion
                                ? {
                                    ...anterior,
                                    prestacion: prestacionActualizada,
                                  }
                                : anterior
                            )
                          }}
                        />
                      )}

                      {coordinacion.prestacion?.estado ===
                        'PENDIENTE_CONTRATACION' && (
                          <div className="adm-detalle-pago">
                            <h4>Estado del agendamiento inicial</h4>

                            {(coordinacion.reservas ?? []).some(
                              (reserva) =>
                                reserva.estado === 'ACTIVA' &&
                                reserva.disponibilidad?.estado === 'RESERVADA'
                            ) ? (
                              <p>
                                Existe una reserva activa. Antes de agendar
                                la primera reunión debe confirmarse que
                                el pago de esta prestación esté RETENIDO.
                                El backend verificará también el horario
                                y las demás condiciones.
                              </p>
                            ) : (
                              <p>
                                La primera reunión no puede agendarse
                                todavía. Se requiere pago verificado y
                                retenido, con reserva ACTIVA y horario
                                RESERVADA.
                              </p>
                            )}
                          </div>
                        )}

                      {coordinacion.prestacion?.estado === 'CONTRATADA' && (
                        <div className="adm-detalle-pago">
                          <h4>Contratación confirmada</h4>
                          <p>
                            La prestación figura CONTRATADA.
                            Revisa las reuniones registradas para consultar
                            su agendamiento.
                          </p>
                        </div>
                      )}


                      {coordinacion.prestacion?.estado ===
                        'PENDIENTE_CONTRATACION' &&
                        pagos.some((pago) =>
                          Number(pago.id_prestacion) ===
                            Number(coordinacion.prestacion.id_prestacion) &&
                          pago.estado_pago === 'RETENIDO' &&
                          pago.estado_verificacion === 'VERIFICADO'
                        ) &&
                        (coordinacion.reservas ?? []).some(
                          (reserva) =>
                            reserva.estado === 'ACTIVA' &&
                            reserva.disponibilidad?.estado === 'RESERVADA'
                        ) &&
                        !(coordinacion.reuniones ?? []).some(
                          (reunion) => reunion.estado === 'AGENDADA'
                        ) && (
                          <button
                            type="button"
                            className="adm-accion"
                            disabled={agendandoPrimeraReunion}
                            onClick={agendarPrimeraReunion}
                          >
                            {agendandoPrimeraReunion
                              ? 'Agendando primera reunión...'
                              : 'Agendar primera reunión'}
                          </button>
                        )}

                      {(coordinacion.reservas ?? []).map((reserva) => (
                        <p key={reserva.id_reserva}>
                          Reserva #{reserva.id_reserva}:
                          {' '}{reserva.estado}
                        </p>
                      ))}

                      {coordinacion.prestacion?.id_prestacion != null && (
                        <SeguimientoPrestacionAdministracion
                          key={coordinacion.prestacion.id_prestacion}
                          idPrestacion={coordinacion.prestacion.id_prestacion}
                        />
                      )}

                      {!coordinacion.prestacion && (
                        <p className="adm-nota">
                          La solicitud todavía no posee una prestación
                          preparada. No hay contratación confirmada.
                        </p>
                      )}
                    </>
                  )}
                </section>

                <h3>Objetivo</h3>
                <p>{seleccion.objetivo}</p>
                <h3>Necesidad empresarial</h3>
                <p>{seleccion.descripcion_necesidad}</p>
                {!asignaciones[seleccion.id_solicitud]?.actual &&
                  !asignaciones[seleccion.id_solicitud]?.error &&
                  !cerrada(seleccion) && (
                    <button
                      type="button"
                      className="adm-accion"
                      disabled={asignando}
                      onClick={() => autoasignar(seleccion)}
                    >
                      {asignando ? 'Asignando...' : 'Autoasignarme'}
                    </button>
                  )}
                <p className="adm-nota">
                  La autoasignación requiere confirmación y permisos
                  administrativos. No confirma pagos ni contrataciones.
                </p>
              </section>
              </VentanaAdministracion>
            )}
          </>
        )}
      </main>
    </div>
  )
}

export default Administracion
